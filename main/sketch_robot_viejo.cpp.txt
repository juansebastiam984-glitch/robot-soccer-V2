struct RegistroPersistente {
  uint32_t version;
  uint8_t numTipos;
  uint8_t maxEventos;
  ContadorError contadores[NUM_TIPOS];
  EventoError eventos[MAX_EVENTOS];
  int32_t totalEventos;
  int32_t indiceEvento;
};

void registrarError(uint8_t tipo) {
  if (tipo >= NUM_TIPOS) return;
  uint32_t ahora = millis();

  portENTER_CRITICAL(&muxErrores);
  if (contadores[tipo].cantidad == 0) {
    contadores[tipo].arranquePrimera = arranqueActual;
    contadores[tipo].tPrimera = ahora;
  }
  contadores[tipo].cantidad++;
  contadores[tipo].arranqueUltima = arranqueActual;
  contadores[tipo].tUltima = ahora;

  if (!logueadoEnEsteArranque[tipo] || ahora - ultimoLogTipo[tipo] >= 1000) {
    logueadoEnEsteArranque[tipo] = true;
    ultimoLogTipo[tipo] = ahora;
    eventos[indiceEvento].arranque = arranqueActual;
    eventos[indiceEvento].tiempo = ahora;
    eventos[indiceEvento].tipo = tipo;
    indiceEvento = (indiceEvento + 1) % MAX_EVENTOS;
    if (totalEventos < MAX_EVENTOS) totalEventos++;
  }
  erroresSinGuardar = true;
  portEXIT_CRITICAL(&muxErrores);
}

void guardarErrores() {
  if (!nvsListo) return;

  RegistroPersistente r;
  memset(&r, 0, sizeof(r));
  r.version = VERSION_REGISTRO;
  r.numTipos = NUM_TIPOS;
  r.maxEventos = MAX_EVENTOS;

  portENTER_CRITICAL(&muxErrores);
  memcpy(r.contadores, contadores, sizeof(contadores));
  memcpy(r.eventos, eventos, sizeof(eventos));
  r.totalEventos = totalEventos;
  r.indiceEvento = indiceEvento;
  erroresSinGuardar = false;
  portEXIT_CRITICAL(&muxErrores);

  size_t escritos = memoria.putBytes("registro", &r, sizeof(r));
  ultimoGuardado = millis();
  if (escritos != sizeof(r)) {
    erroresSinGuardar = true;
    Serial.println("ADVERTENCIA: no se pudo guardar el registro en la memoria permanente.");
  }
}

void cargarErrores() {
  nvsListo = memoria.begin("errores", false);
  if (!nvsListo) {
    Serial.println("ADVERTENCIA: no se pudo abrir la memoria permanente. Los errores NO se guardaran.");
    return;
  }

  arranqueActual = memoria.getUInt("arranques", 0) + 1;
  memoria.putUInt("arranques", arranqueActual);

  if (!memoria.isKey("registro")) return;

  size_t largo = memoria.getBytesLength("registro");
  if (largo == 0) return;

  RegistroPersistente r;
  if (largo != sizeof(r)) {
    memoria.remove("registro");
    return;
  }
  memoria.getBytes("registro", &r, sizeof(r));

  bool valido = (r.version == VERSION_REGISTRO && r.numTipos == NUM_TIPOS &&
                 r.maxEventos == MAX_EVENTOS &&
                 r.totalEventos >= 0 && r.totalEventos <= MAX_EVENTOS &&
                 r.indiceEvento >= 0 && r.indiceEvento < MAX_EVENTOS);
  if (!valido) {
    memoria.remove("registro");
    return;
  }

  memcpy(contadores, r.contadores, sizeof(contadores));
  memcpy(eventos, r.eventos, sizeof(eventos));
  totalEventos = r.totalEventos;
  indiceEvento = r.indiceEvento;

  erroresPrevios = 0;
  for (int t = 0; t < NUM_TIPOS; t++) {
    if (!esEvento(t)) erroresPrevios += contadores[t].cantidad;
  }
}

String formatoTiempo(unsigned long ms) {
  unsigned long s = ms / 1000;
  if (s < 60) return String(ms / 1000.0, 1) + "s";
  return String(s / 60) + "m " + String(s % 60) + "s";
}

String formatoMomento(uint32_t arranque, uint32_t ms) {
  String s = "arranque #" + String(arranque) + ", " + formatoTiempo(ms);
  if (arranque == arranqueActual) s += " (actual)";
  return s;
}

void mostrarErrores() {
  ContadorError copiaC[NUM_TIPOS];
  EventoError copiaE[MAX_EVENTOS];
  int n, idx;

  portENTER_CRITICAL(&muxErrores);
  memcpy(copiaC, contadores, sizeof(contadores));
  memcpy(copiaE, eventos, sizeof(eventos));
  n = totalEventos;
  idx = indiceEvento;
  portEXIT_CRITICAL(&muxErrores);

  Serial.println();
  Serial.println("===== ERRORES REGISTRADOS =====");
  Serial.print("Arranque actual: #");
  Serial.print(arranqueActual);
  Serial.print(" | tiempo encendido: ");
  Serial.println(formatoTiempo(millis()));
  if (!nvsListo) {
    Serial.println("ADVERTENCIA: memoria permanente no disponible, estos errores se pierden al reiniciar.");
  }

  bool hay = false;
  bool hayError = false;
  for (int t = 0; t < NUM_TIPOS; t++) {
    if (copiaC[t].cantidad == 0) continue;
    hay = true;
    if (!esEvento(t)) hayError = true;
    Serial.print("- ");
    Serial.print(NOMBRES_ERROR[t]);
    Serial.print(": ");
    Serial.print(copiaC[t].cantidad);
    Serial.print(" vez/veces (primera: ");
    Serial.print(formatoMomento(copiaC[t].arranquePrimera, copiaC[t].tPrimera));
    Serial.print(" | ultima: ");
    Serial.print(formatoMomento(copiaC[t].arranqueUltima, copiaC[t].tUltima));
    Serial.println(")");
  }

  if (!hay) {
    Serial.println("Sin errores registrados.");
    Serial.println("===============================");
    return;
  }

  if (!hayError) Serial.println("(Sin errores: solo hay eventos de conexion)");
  Serial.println("Ultimos eventos (el mas reciente al final):");
  int inicio = (n < MAX_EVENTOS) ? 0 : idx;
  for (int i = 0; i < n; i++) {
    EventoError &e = copiaE[(inicio + i) % MAX_EVENTOS];
    if (e.tipo >= NUM_TIPOS) continue;
    Serial.print("  [");
    Serial.print(formatoMomento(e.arranque, e.tiempo));
    Serial.print("] ");
    Serial.println(NOMBRES_ERROR[e.tipo]);
  }
  Serial.println("Escribe 'limpiar errores' para borrarlos (tambien de la memoria permanente).");
  Serial.println("===============================");
}

const uint32_t VERSION_AJUSTES = 3;

bool guardarAjustes();
extern int pwmMinimo;

void cargarAjustes() {
  ajustesListos = ajustes.begin("ajustes", false);
  if (!ajustesListos) return;
  bool versionNueva = (ajustes.getUInt("version", 0) != VERSION_AJUSTES);
  if (!versionNueva) {
    if (ajustes.isKey("kp")) kp = ajustes.getFloat("kp");
    if (ajustes.isKey("kd")) kd = ajustes.getFloat("kd");
  }
  if (ajustes.isKey("ki")) ki = ajustes.getFloat("ki");
  // Si el valor guardado esta corrupto (NaN o fuera de rango), vuelve al valor por defecto.
  if (!(kp >= 0 && kp <= 20)) kp = 4.0;
  if (!(kd >= 0 && kd <= 20)) kd = 0.30;
  if (!(ki >= 0 && ki <= 20)) ki = 2.0;
  if (ajustes.isKey("invertir")) signoGiro = ajustes.getInt("invertir");
  if (signoGiro != 1 && signoGiro != -1) signoGiro = 1;
  if (ajustes.isKey("pwmmin")) pwmMinimo = constrain(ajustes.getInt("pwmmin"), 0, 150);
  if (ajustes.isKey("trim")) {
    trimMotor = ajustes.getFloat("trim");
    if (!(trimMotor >= -40 && trimMotor <= 40)) trimMotor = 0;
  }
  if (ajustes.isKey("volumen")) buzzerVolumen = constrain(ajustes.getInt("volumen"), 0, BUZZER_VOLUMEN_MAX);
  if (versionNueva) {
    ajustes.putUInt("version", VERSION_AJUSTES);
    guardarAjustes();
  }
}

bool guardarAjustes() {
  if (!ajustesListos) return false;
  bool ok = ajustes.putFloat("kp", kp) == sizeof(float);
  ok &= ajustes.putFloat("kd", kd) == sizeof(float);
  ok &= ajustes.putFloat("ki", ki) == sizeof(float);
  ok &= ajustes.putInt("invertir", signoGiro) == sizeof(int32_t);
  ok &= ajustes.putInt("pwmmin", pwmMinimo) == sizeof(int32_t);
  ok &= ajustes.putInt("volumen", buzzerVolumen) == sizeof(int32_t);
  ok &= ajustes.putFloat("trim", trimMotor) == sizeof(float);
  return ok;
}

float diferenciaAngular(float actual, float anterior) {
  float diferencia = actual - anterior;
  while (diferencia > 180) diferencia -= 360;
  while (diferencia < -180) diferencia += 360;
  return diferencia;
}

int16_t leer16() {
  int16_t alto = Wire.read();
  int16_t bajo = Wire.read();
  return (int16_t)((alto << 8) | bajo);
}

bool escribirRegistro(uint8_t registro, uint8_t valor) {
  Wire.beginTransmission(mpuAddr);
  Wire.write(registro);
  Wire.write(valor);
  return Wire.endTransmission() == 0;
}

bool buscarMPU() {
  uint8_t candidatas[2] = {mpuAddr, (uint8_t)(mpuAddr == 0x68 ? 0x69 : 0x68)};
  for (int i = 0; i < 2; i++) {
    Wire.beginTransmission(candidatas[i]);
    if (Wire.endTransmission() == 0) {
      mpuAddr = candidatas[i];
      return true;
    }
  }
  return false;
}

bool iniciarMPU() {
  if (!buscarMPU()) return false;
  escribirRegistro(0x6B, 0x01);   // despierta, reloj del giroscopio X
  escribirRegistro(0x1A, 0x03);   // filtro pasa-bajos ~42 Hz (menos ruido en el termino D)
  escribirRegistro(0x1B, 0x08);   // giroscopio +-500 grados/s
  escribirRegistro(0x1C, 0x08);   // acelerometro +-4 g (soporta mejor los choques)
  return true;
}

void iniciarI2C() {
  Wire.begin(PIN_SDA, PIN_SCL);
  Wire.setClock(100000);
  Wire.setTimeOut(10);   // ms: un bus colgado no debe bloquear el loop 50 ms por lectura
}

// Libera un bus I2C colgado (esclavo reteniendo SDA): 9 pulsos de reloj + condicion de parada.
void recuperarI2C() {
  Wire.end();
  pinMode(PIN_SDA, INPUT_PULLUP);
  pinMode(PIN_SCL, OUTPUT);
  digitalWrite(PIN_SCL, HIGH);
  for (int i = 0; i < 9; i++) {
    digitalWrite(PIN_SCL, LOW);
    delayMicroseconds(5);
    digitalWrite(PIN_SCL, HIGH);
    delayMicroseconds(5);
  }
  pinMode(PIN_SDA, OUTPUT);
  digitalWrite(PIN_SDA, LOW);
  delayMicroseconds(5);
  digitalWrite(PIN_SCL, HIGH);
  delayMicroseconds(5);
  digitalWrite(PIN_SDA, HIGH);
  delayMicroseconds(5);
  iniciarI2C();
}

int leerID() {
  Wire.beginTransmission(mpuAddr);
  Wire.write(0x75);
  if (Wire.endTransmission(false) != 0) return -1;
  if (Wire.requestFrom((uint16_t)mpuAddr, (uint8_t)1, true) != 1) return -1;
  return Wire.read();
}

int leerMPU() {
  Wire.beginTransmission(mpuAddr);
  Wire.write(0x3B);
  if (Wire.endTransmission(false) != 0) return 1;
  if (Wire.requestFrom((uint16_t)mpuAddr, (uint8_t)14, true) != 14) return 1;

  ax = leer16();
  ay = leer16();
  az = leer16();
  leer16();
  gx = leer16();
  gy = leer16();
  gz = leer16();

  if (ax == 0 && ay == 0 && az == 0 && gx == 0 && gy == 0 && gz == 0) return 2;
  return 0;
}

int ultimoPWMA = 0, ultimoPWMB = 0;

int pwmMinimo = 40;

// Equilibrio entre motores, en %: positivo quita potencia al motor A (izquierdo),
// negativo quita potencia al motor B (derecho). Corrige que el carro se vaya hacia un lado.
float trimMotor = 0;

int aplicarTrim(int v, bool quitar) {
  if (!quitar) return v;
  return (int)(v * (1.0f - fabsf(trimMotor) / 100.0f));
}

int compensarZonaMuerta(int v) {
  if (v == 0 || pwmMinimo <= 0) return v;
  int s = (v > 0) ? 1 : -1;
  int a = abs(v);
  if (a > 255) a = 255;
  return s * (pwmMinimo + (a * (255 - pwmMinimo)) / 255);
}

void motorA(int velocidad) {
  ultimoPWMA = velocidad;
  velocidad = compensarZonaMuerta(aplicarTrim(velocidad, trimMotor > 0));
  if (velocidad >= 0) {
    digitalWrite(pinAIN1, HIGH);
    digitalWrite(pinAIN2, LOW);
  } else {
    digitalWrite(pinAIN1, LOW);
    digitalWrite(pinAIN2, HIGH);
    velocidad = -velocidad;
  }
  PWM_ESCRIBIR(pinPWMA, velocidad);
}

void motorB(int velocidad) {
  ultimoPWMB = velocidad;
  velocidad = compensarZonaMuerta(aplicarTrim(velocidad, trimMotor < 0));
  if (velocidad >= 0) {
    digitalWrite(pinBIN1, HIGH);
    digitalWrite(pinBIN2, LOW);
  } else {
    digitalWrite(pinBIN1, LOW);
    digitalWrite(pinBIN2, HIGH);
    velocidad = -velocidad;
  }
  PWM_ESCRIBIR(pinPWMB, velocidad);
}

void detenerMotores() {
  motorA(0);
  motorB(0);
}

void aplicarMovimiento(float corr) {
  int base = velocidadBase;
  velocidadAplicada = base;

  switch (movimiento) {
    case ADELANTE:
    case ATRAS: {
      int dir = (movimiento == ADELANTE) ? 1 : -1;

      if (direccionManual) {

        float izqF = dir * base - corr;
        float derF = dir * base + corr;
        if (derF > 255) { izqF -= derF - 255; derF = 255; }
        else if (derF < -255) { izqF += -255 - derF; derF = -255; }
        if (izqF > 255) { derF -= izqF - 255; izqF = 255; }
        else if (izqF < -255) { derF += -255 - izqF; izqF = -255; }
        motorA(constrain((int)izqF, -255, 255));
        motorB(constrain((int)derF, -255, 255));
        break;
      }

      int limite = (CORRECCION_MAXIMA < base) ? CORRECCION_MAXIMA : base;
      int c = constrain((int)corr, -limite, limite);
      int izq = dir * base - c;
      int der = dir * base + c;
      if (der > 255) { izq -= der - 255; der = 255; }
      else if (der < -255) { izq += -255 - der; der = -255; }
      if (izq > 255) { der -= izq - 255; izq = 255; }
      else if (izq < -255) { der += -255 - izq; izq = -255; }
      motorA(izq);
      motorB(der);
      break;
    }
    case GIRO_IZQ: {
      int g = base * GIRO_PORCENTAJE / 100;
      motorA(-g);
      motorB(g);
      break;
    }
    case GIRO_DER: {
      int g = base * GIRO_PORCENTAJE / 100;
      motorA(g);
      motorB(-g);
      break;
    }
    default:
      detenerMotores();
      break;
  }
}

void fijarRumbo() {
  rumboObjetivo = anguloZ;
  ultimaCorreccion = 0;
  integralRumbo = 0;
  corrigiendo = rumboHabilitado && mpuListo && !primeraLectura && giroCalibrado;
}

void apagarCorreccion() {
  if (!corrigiendo) return;
  corrigiendo = false;
  ultimaCorreccion = 0;
  aplicarMovimiento(0);
}

void actualizarMovimiento() {
  Movimiento nuevo = orden;

  if (frenando) {
    if (nuevo != PARADO) {
      frenando = false;
    } else if (millis() >= frenoHasta) {
      frenando = false;
      detenerMotores();
      return;
    } else {
      return;
    }
  }

  if (nuevo != movimiento) {

    Movimiento previo = movimiento;
    if (nuevo == PARADO && FRENO_ACTIVO_MS > 0 && !cortePorFaltaDeDatos &&
        (abs(ultimoPWMA) >= FRENO_ACTIVO_MIN || abs(ultimoPWMB) >= FRENO_ACTIVO_MIN)) {
      int fa = (abs(ultimoPWMA) >= FRENO_ACTIVO_MIN)
                 ? constrain(abs(ultimoPWMA) * FRENO_ACTIVO_PORC / 100, 0, FRENO_ACTIVO_MAX) : 0;
      int fb = (abs(ultimoPWMB) >= FRENO_ACTIVO_MIN)
                 ? constrain(abs(ultimoPWMB) * FRENO_ACTIVO_PORC / 100, 0, FRENO_ACTIVO_MAX) : 0;
      movimiento = PARADO;
      esperandoRelock = false;
      corrigiendo = false;
      ultimaCorreccion = 0;
      motorA(ultimoPWMA > 0 ? -fa : fa);
      motorB(ultimoPWMB > 0 ? -fb : fb);
      frenando = true;
      frenoHasta = millis() + FRENO_ACTIVO_MS;
      return;
    }
    movimiento = nuevo;
    esperandoRelock = false;
    if (movimiento == ADELANTE || movimiento == ATRAS) {
      if (previo == GIRO_IZQ || previo == GIRO_DER) {
        // Venia girando sobre su eje: espera a que se asiente antes de fijar el rumbo.
        corrigiendo = false;
        ultimaCorreccion = 0;
        esperandoRelock = true;
        finManual = millis();
      } else {
        fijarRumbo();
      }
    } else {
      corrigiendo = false;
      ultimaCorreccion = 0;
    }
    aplicarMovimiento(direccionManual ? corrManual : ultimaCorreccion);
    return;
  }

  if (movimiento == ADELANTE || movimiento == ATRAS) {
    if (direccionManual) {

      corrigiendo = false;
      rumboObjetivo = anguloZ;
      ultimaCorreccion = 0;
      integralRumbo = 0;
      esperandoRelock = true;
      finManual = millis();
      aplicarMovimiento(corrManual);
    } else if (esperandoRelock) {

      ultimaCorreccion = (mpuListo && giroCalibrado) ? -KD_FRENO_GIRO * velAngularZ : 0;
      aplicarMovimiento(ultimaCorreccion);
      if (millis() - finManual >= RETARDO_RELOCK) {
        esperandoRelock = false;
        ultimaCorreccion = 0;
        aplicarMovimiento(0);
        if (rumboHabilitado && mpuListo && giroCalibrado) fijarRumbo();
      }
    } else {
      if (!corrigiendo && rumboHabilitado && mpuListo && giroCalibrado) {
        fijarRumbo();
      }
      if (velocidadBase != velocidadAplicada) {
        aplicarMovimiento(ultimaCorreccion);
      }
    }
  } else if (velocidadBase != velocidadAplicada) {
    aplicarMovimiento(ultimaCorreccion);
  }
}

void corregirRumbo(float giro, float dt) {
  if (movimiento != ADELANTE && movimiento != ATRAS) return;
  if (direccionManual) return;

  float error = diferenciaAngular(rumboObjetivo, anguloZ);

  if (abs(error) > DESVIO_MAXIMO) {
    rumboObjetivo = anguloZ;
    error = 0;
    integralRumbo = 0;
    Serial.println("Desvio mayor a 45 grados: nuevo rumbo.");
  }

  if (error > ZONA_MUERTA_RUMBO) error -= ZONA_MUERTA_RUMBO;
  else if (error < -ZONA_MUERTA_RUMBO) error += ZONA_MUERTA_RUMBO;
  else error = 0;

  if (abs(error) < 10.0 && ki > 0) {
    integralRumbo += error * dt;
    float tope = INTEGRAL_MAX_PWM / ki;
    integralRumbo = constrain(integralRumbo, -tope, tope);
  }

  ultimaCorreccion = kp * error + ki * integralRumbo - kd * giro;
  aplicarMovimiento(ultimaCorreccion);
}

float inclinacionPlaca() {
  float c = fabs(arribaZ);
  if (c > 1) c = 1;
  return acos(c) * 180.0 / PI;
}

bool calibrarGiroscopio() {
  if (!mpuListo) {
    Serial.println("Calibracion cancelada: MPU6050 no disponible.");
    return false;
  }

  orden = PARADO;
  movimiento = PARADO;
  corrigiendo = false;
  detenerMotores();

  Serial.println("Calibrando: deja el robot quieto y apoyado en el piso (2 segundos)...");
  double sumaA[3] = {0, 0, 0};
  double sumaG[3] = {0, 0, 0};
  int16_t minimo[3] = {32767, 32767, 32767};
  int16_t maximo[3] = {-32768, -32768, -32768};
  int validas = 0;
  for (int i = 0; i < 200; i++) {
    if (leerMPU() == 0) {
      int16_t a[3] = {ax, ay, az};
      int16_t g[3] = {gx, gy, gz};
      for (int k = 0; k < 3; k++) {
        sumaA[k] += a[k];
        sumaG[k] += g[k];
        if (g[k] < minimo[k]) minimo[k] = g[k];
        if (g[k] > maximo[k]) maximo[k] = g[k];
      }
      validas++;
    }
    delay(10);
  }

  if (validas < 150) {
    Serial.println("Calibracion fallida: pocas lecturas buenas del sensor.");
    return false;
  }
  for (int k = 0; k < 3; k++) {
    if (maximo[k] - minimo[k] > 500) {
      Serial.println("Calibracion descartada: el robot se movio. Intenta de nuevo.");
      return false;
    }
  }

  float mx = sumaA[0] / validas;
  float my = sumaA[1] / validas;
  float mz = sumaA[2] / validas;
  float modulo = sqrt(mx * mx + my * my + mz * mz);
  if (modulo < LSB_POR_G * 0.73 || modulo > LSB_POR_G * 1.28) {
    Serial.println("Calibracion descartada: la aceleracion no es 1 G (el robot se mueve o el sensor falla).");
    return false;
  }

  gxOffset = sumaG[0] / validas;
  gyOffset = sumaG[1] / validas;
  gzOffset = sumaG[2] / validas;
  arribaX = mx / modulo;
  arribaY = my / modulo;
  arribaZ = mz / modulo;
  giroCalibrado = true;
  primeraLectura = true;

  Serial.print("Calibracion lista. Placa inclinada ");
  Serial.print(inclinacionPlaca(), 1);
  Serial.println(arribaZ < 0 ? " grados (boca abajo)." : " grados (boca arriba).");
  imprimirBloqueCalibracion();
  return true;
}

#define FILTRAR_MAC

ControllerPtr miMando = nullptr;

volatile bool mandoListo = false;

volatile unsigned long ultimoDatoMando = 0;

bool cortePorFaltaDeDatos = false;

unsigned long reportesMando = 0;
unsigned long huecoMaximoMando = 0;
unsigned long huecosLargos = 0;
unsigned long cortesMando = 0;
unsigned long inicioEstadisticas = 0;

const int ZONA_MUERTA_STICK = 40;
const int SIGNO_STICK = 1;

const int UMBRAL_STICK_CORRECTOR = 25;
const int HISTERESIS_STICK = 8;

bool stickTocado = false;
bool centroStickListo = false;
float sumaCentroX = 0;
int muestrasCentroX = 0;

void onConnectedController(ControllerPtr ctl) {
  if (miMando != nullptr) {
    Serial.println("Ya hay un mando conectado; se ignora este.");
    ctl->disconnect();
    return;
  }

  ControllerProperties props = ctl->getProperties();

#ifdef FILTRAR_MAC
  bool coincide = true;
  for (int i = 0; i < 6; i++) {
    if (props.btaddr[i] != MAC_MANDO[i]) { coincide = false; break; }
  }
  if (!coincide) {
    Serial.println("Mando con MAC distinta a la autorizada: desconectado.");
    ctl->disconnect();
    return;
  }
#endif

  miMando = ctl;
  mandoListo = false;
  stickTocado = false;
  centroStickListo = false;
  centroStickX = 0;
  sumaCentroX = 0;
  muestrasCentroX = 0;
  ultimoDatoMando = millis();
  reportesMando = 0;
  huecoMaximoMando = 0;
  huecosLargos = 0;
  cortesMando = 0;
  inicioEstadisticas = millis();
  buscarMandosNuevos(false);
  pitar(DURACION_CONEXION);
  registrarError(EVT_MANDO_CONECTADO);
  Serial.printf("Mando conectado: %s  MAC=%02X:%02X:%02X:%02X:%02X:%02X\n",
    ctl->getModelName().c_str(),
    props.btaddr[0], props.btaddr[1], props.btaddr[2],
    props.btaddr[3], props.btaddr[4], props.btaddr[5]);
  Serial.println("Suelta los gatillos y centra el stick para habilitar el movimiento...");
}

void onDisconnectedController(ControllerPtr ctl) {
  if (miMando == ctl) {
    miMando = nullptr;
    mandoListo = false;
    orden = PARADO;
    buscarMandosNuevos(true);
    registrarError(EVT_MANDO_DESCONECTADO);
    Serial.println("Mando desconectado: motores frenados. Presiona PS en el mando para reconectar.");
  }
}

void leerMando() {
  if (!miMando || !miMando->isConnected()) {
    orden = PARADO;
    return;
  }

  unsigned long ahoraMs = millis();
  if (miMando->hasData()) {
    unsigned long hueco = ahoraMs - ultimoDatoMando;
    if (reportesMando > 0) {
      if (hueco > huecoMaximoMando) huecoMaximoMando = hueco;
      if (hueco > 100) huecosLargos++;
    }
    if (cortePorFaltaDeDatos) {
      Serial.printf("Bluetooth: volvieron los datos tras %lu ms sin nada\n", hueco);
      cortePorFaltaDeDatos = false;
    }
    reportesMando++;
    ultimoDatoMando = ahoraMs;
  }

  if (ahoraMs - ultimoDatoMando > TIEMPO_MAX_SIN_DATOS) {
    if (!cortePorFaltaDeDatos) {
      cortePorFaltaDeDatos = true;
      cortesMando++;
      Serial.printf("Bluetooth: %lu ms sin datos del mando, robot frenado\n", TIEMPO_MAX_SIN_DATOS);
    }
    orden = PARADO;
    return;
  }

  int32_t r2 = miMando->throttle();
  int32_t l2 = miMando->brake();
  int32_t ejeCrudo = miMando->axisX();

  if (!centroStickListo && miMando->hasData()) {
    if (abs(ejeCrudo) < 150) {
      sumaCentroX += ejeCrudo;
      muestrasCentroX++;
    } else {
      sumaCentroX = 0;
      muestrasCentroX = 0;
    }
    if (muestrasCentroX >= 30) {
      centroStickX = sumaCentroX / muestrasCentroX;
      centroStickListo = true;
      Serial.printf("Centro del stick medido: %.0f\n", centroStickX);
    }
  }
  int32_t ejeX = (int32_t)((ejeCrudo - centroStickX) * SIGNO_STICK);

  if (!mandoListo) {
    if (r2 < ZONA_MUERTA_GATILLO && l2 < ZONA_MUERTA_GATILLO && abs(ejeX) < ZONA_MUERTA_STICK) {
      mandoListo = true;
      Serial.println("Mando listo: movimiento habilitado.");
    } else {
      orden = PARADO;
      return;
    }
  }

  int velocidad = map(r2, 0, 1023, 0, 255) - map(l2, 0, 1023, 0, 255);
  velocidad = constrain(velocidad, -255, 255);
  bool girando = abs(ejeX) > ZONA_MUERTA_STICK;

  float giroNorm = 0;
  if (girando) {
    float x = ((float)abs(ejeX) - ZONA_MUERTA_STICK) / (512.0 - ZONA_MUERTA_STICK);
    x = constrain(x, 0.0f, 1.0f);
    x = powf(x, CURVA_GIRO);
    giroNorm = (ejeX > 0) ? x : -x;
  }

  if (abs(velocidad) < 15 && girando) {

    orden = (ejeX > 0) ? GIRO_DER : GIRO_IZQ;
    velocidadBase = 120 + (int)(fabsf(giroNorm) * (255 - 120));
    direccionManual = false;
  } else if (abs(velocidad) >= 15) {
    orden = (velocidad > 0) ? ADELANTE : ATRAS;
    velocidadBase = abs(velocidad);

    if (!stickTocado && abs(ejeX) > UMBRAL_STICK_CORRECTOR) stickTocado = true;
    else if (stickTocado && abs(ejeX) < UMBRAL_STICK_CORRECTOR - HISTERESIS_STICK) stickTocado = false;

    if (stickTocado) {

      direccionManual = true;

      corrManual = -giroNorm * GIRO_MAX_MOVIENDO;
    } else {
      direccionManual = false;
    }
  } else {
    orden = PARADO;
    direccionManual = false;
  }

  if (DIAGNOSTICO_ORDEN) {
    static Movimiento ordenAnterior = PARADO;
    if (orden != ordenAnterior) {
      const char* nombres[] = {"PARADO", "ADELANTE", "ATRAS", "GIRO_IZQ", "GIRO_DER"};
      Serial.printf("[%lu ms] orden=%s r2=%ld l2=%ld x=%ld (ms desde ultimo dato del mando: %lu)\n",
        millis(), nombres[orden], (long)r2, (long)l2, (long)ejeX,
        (unsigned long)(millis() - ultimoDatoMando));
      ordenAnterior = orden;
    }

    static bool manualAnterior = false;
    if (direccionManual != manualAnterior) {
      Serial.printf("[%lu ms] stick %s  x=%ld (ms desde ultimo dato del mando: %lu)\n",
        millis(), direccionManual ? "TOCADO" : "SOLTADO", (long)ejeX,
        (unsigned long)(millis() - ultimoDatoMando));
      manualAnterior = direccionManual;
    }
  }
}

const bool MOSTRAR_BOTONES = true;

void mostrarBotones() {
  if (!MOSTRAR_BOTONES || !miMando || !miMando->isConnected()) return;

  struct Boton { const char* nombre; bool estado; };
  uint8_t d = miMando->dpad();
  Boton lista[] = {
    {"X (Cruz)",        miMando->a()},
    {"O (Circulo)",     miMando->b()},
    {"Cuadrado",        miMando->x()},
    {"Triangulo",       miMando->y()},
    {"L1",              miMando->l1()},
    {"R1",              miMando->r1()},
    {"L3 (stick izq)",  miMando->thumbL()},
    {"R3 (stick der)",  miMando->thumbR()},
    {"Flecha arriba",   (d & 0x01) != 0},
    {"Flecha abajo",    (d & 0x02) != 0},
    {"Flecha derecha",  (d & 0x04) != 0},
    {"Flecha izquierda",(d & 0x08) != 0},
    {"SHARE",           miMando->miscSelect()},
    {"OPTIONS",         miMando->miscStart()},
    {"PS",              miMando->miscSystem()},
  };
  const int n = sizeof(lista) / sizeof(lista[0]);

  for (int i = 0; i < n; i++) {

      Serial.printf("Boton %s %s\n", lista[i].nombre,
                    lista[i].estado ? "presionado" : "soltado");
      anterior[i] = lista[i].estado;
    }
  }
}

const float TEMP_ALERTA_C = 70.0;
const float TEMP_MUY_ALTA_C = 80.0;
const unsigned long PERIODO_PITIDO_ALERTA = 1000;
const unsigned long DURACION_PITIDO_ALERTA = 150;
const unsigned long INTERVALO_TEMP = 1000;
bool alertaTemperatura = false;
bool alertaMuyAlta = false;
float ultimaTempC = NAN;
unsigned long ultimaLecturaTemp = 0;

void revisarTemperatura(unsigned long ahora) {
  if (ahora - ultimaLecturaTemp < INTERVALO_TEMP) return;
  ultimaLecturaTemp = ahora;

  float temp = temperatureRead();
  ultimaTempC = temp;
  if (isnan(temp)) return;

  if (!alertaTemperatura && temp > TEMP_ALERTA_C) {
    alertaTemperatura = true;
    Serial.printf("ALERTA: procesador a %.1f C\n", temp);
  } else if (alertaTemperatura && temp < TEMP_ALERTA_C - 5) {
    alertaTemperatura = false;
    Serial.printf("Temperatura normal otra vez: %.1f C\n", temp);
  }

  if (!alertaMuyAlta && temp > TEMP_MUY_ALTA_C) {
    alertaMuyAlta = true;
    Serial.printf("ALERTA GRAVE: procesador a %.1f C, apaga el robot\n", temp);
  } else if (alertaMuyAlta && temp < TEMP_MUY_ALTA_C - 3) {
    alertaMuyAlta = false;
  }
}

void buzzerEncender(bool encender) {
  static bool encendido = false;
  if (encendido == encender) return;
  encendido = encender;
  BUZZER_ESCRIBIR(encender ? constrain(buzzerVolumen, 0, BUZZER_VOLUMEN_MAX) : 0);
}

void actualizarBuzzer(unsigned long ahora) {
  revisarTemperatura(ahora);

  bool sonar = false;
  if ((long)(buzzerHasta - ahora) > 0) {
    sonar = true;
  } else if (alertaMuyAlta) {
    sonar = true;
  } else if (alertaTemperatura) {
    sonar = (ahora % PERIODO_PITIDO_ALERTA) < DURACION_PITIDO_ALERTA;
  } else if (miMando == nullptr) {
    sonar = (ahora % (2 * DURACION_BUSCANDO)) < DURACION_BUSCANDO;
  }
  buzzerEncender(sonar);
}

void ajustarValor(String comando, const char* nombre, float &valor) {
  String arg = comando.substring(2);
  arg.trim();
  arg.replace(',', '.');

  if (arg.length() == 0) {
    Serial.print(nombre);
    Serial.print(" = ");
    Serial.println(valor, 2);
    return;
  }

  float nuevo = arg.toFloat();
  bool esNumero = (nuevo != 0 || arg.charAt(0) == '0');
  if (!esNumero || nuevo < 0 || nuevo > 20) {
    Serial.println("Valor invalido. Usa un numero entre 0 y 20, ej: kp 3");
    return;
  }

  valor = nuevo;
  bool guardado = guardarAjustes();
  Serial.print(nombre);
  Serial.print(" = ");
  Serial.print(valor, 2);
  Serial.println(guardado ? " (guardado)" : " (activo, pero no se pudo guardar)");
}

void procesarComandoSerial(String comando) {
  comando.trim();
  comando.toLowerCase();

  if (comando == "errores") {
    mostrarErrores();
  } else if (comando == "limpiar errores") {
    limpiarErrores();
  } else if (comando == "rumbo") {
    rumboHabilitado = !rumboHabilitado;
    if (!rumboHabilitado) {
      apagarCorreccion();
    } else if (movimiento == ADELANTE || movimiento == ATRAS) {
      fijarRumbo();
    }
    Serial.print("Correccion de rumbo: ");
    Serial.println(rumboHabilitado ? "ACTIVADA" : "DESACTIVADA");
    if (rumboHabilitado && !giroCalibrado) {
      Serial.println("Pero el giroscopio no esta calibrado: no actua hasta que escribas 'calibrar'.");
    }
  } else if (comando == "signo") {
    signoGiro = -signoGiro;
    bool guardado = guardarAjustes();
    if (corrigiendo) rumboObjetivo = anguloZ;
    Serial.print("Inversion del giro: ");
    Serial.print(signoGiro);
    Serial.println(guardado ? " (guardado)" : " (activo, pero no se pudo guardar)");
  } else if (comando.startsWith("kp")) {
    ajustarValor(comando, "kp", kp);
  } else if (comando.startsWith("kd")) {
    ajustarValor(comando, "kd", kd);
  } else if (comando.startsWith("ki")) {
    ajustarValor(comando, "ki", ki);
  } else if (comando == "calibrar") {
    calibrarGiroscopio();
  } else if (comando == "ajustes") {
    mostrarAjustes();
  } else if (comando == "bt") {
    mostrarEstadisticasBT();
  } else if (comando.startsWith("pwmmin")) {
    String arg = comando.substring(6);
    arg.trim();
    if (arg.length() == 0) {
      Serial.printf("pwmmin = %d\n", pwmMinimo);
    } else {
      int v = arg.toInt();
      if ((v == 0 && arg.charAt(0) != '0') || v < 0 || v > 150) {
        Serial.println("Valor invalido. Usa un numero entre 0 y 150, ej: pwmmin 45");
      } else {
        pwmMinimo = v;
        bool guardado = guardarAjustes();
        Serial.printf("pwmmin = %d%s\n", pwmMinimo, guardado ? " (guardado)" : " (activo, pero no se pudo guardar)");
      }
    }
  } else if (comando.startsWith("trim")) {
    String arg = comando.substring(4);
    arg.trim();
    arg.replace(',', '.');
    if (arg.length() == 0) {
      Serial.printf("trim = %.1f (positivo quita potencia al motor izquierdo A, negativo al derecho B)\n", trimMotor);
    } else {
      float v = arg.toFloat();
      bool esNumero = (v != 0 || arg.charAt(0) == '0' || arg.startsWith("-0"));
      if (!esNumero || v < -40 || v > 40) {
        Serial.println("Valor invalido. Usa un numero entre -40 y 40, ej: trim 8");
      } else {
        trimMotor = v;
        bool guardado = guardarAjustes();
        Serial.printf("trim = %.1f%s\n", trimMotor, guardado ? " (guardado)" : " (activo, pero no se pudo guardar)");
      }
    }
  } else if (comando == "bip") {
    pitar(200);
    Serial.println("bip");
  } else if (comando == "temp") {
    if (isnan(temperatureRead())) {
      Serial.println("Este core no tiene sensor de temperatura del procesador: la alerta no funciona.");
    } else {
      Serial.printf("Procesador: %.1f C (alerta a %.0f C, continuo a %.0f C)\n",
                    temperatureRead(), TEMP_ALERTA_C, TEMP_MUY_ALTA_C);
    }
  } else if (comando.startsWith("volumen")) {
    String arg = comando.substring(7);
    arg.trim();
    if (arg.length() == 0) {
      Serial.printf("volumen = %d (de 0 a %d)\n", buzzerVolumen, BUZZER_VOLUMEN_MAX);
    } else {
      int v = arg.toInt();
      if ((v == 0 && arg.charAt(0) != '0') || v < 0 || v > BUZZER_VOLUMEN_MAX) {
        Serial.printf("Valor invalido. Usa un numero entre 0 y %d, ej: volumen 30\n", BUZZER_VOLUMEN_MAX);
      } else {
        buzzerVolumen = v;
        bool guardado = guardarAjustes();
        Serial.printf("volumen = %d%s\n", buzzerVolumen, guardado ? " (guardado)" : " (activo, pero no se pudo guardar)");
        pitar(200);
      }
    }
  } else if (comando.length() > 0) {
    Serial.println("Comando desconocido. Disponibles: errores | limpiar errores | rumbo | kp | kd | ki | signo | calibrar | ajustes | bt | pwmmin | trim | bip | volumen | temp");
  }
}

void actualizarMPU(unsigned long ahora) {
  if (!mpuListo) {
    if (ahora - ultimoReintento >= 3000) {
      ultimoReintento = ahora;
      if (!(iniciarMPU() && leerID() == 0x68)) {
        recuperarI2C();   // por si el bus quedo colgado
      } else {
        mpuListo = true;
        primeraLectura = true;
        erroresI2CSeguidos = 0;
        rechazosSeguidos = 0;
        ultimaLecturaGiro = ahora;
        registrarError(EVT_MPU_RECONECTADO);
        Serial.print("MPU6050 reconectado en 0x");
        Serial.println(mpuAddr, HEX);
      }
    }
    return;
  }

  if (ahora - ultimaLecturaGiro < INTERVALO_LECTURA) return;
  ultimaLecturaGiro = ahora;

  int resultado = leerMPU();
  if (resultado != 0) {
    registrarError(resultado == 1 ? ERR_I2C_LECTURA : ERR_LECTURA_VACIA);
    erroresI2CSeguidos++;
    if (erroresI2CSeguidos == 3) {
      uint8_t anterior = mpuAddr;
      if (iniciarMPU() && mpuAddr != anterior) {
        erroresI2CSeguidos = 0;
        registrarError(EVT_MPU_RECONECTADO);
        Serial.print("MPU6050 cambio de direccion a 0x");
        Serial.println(mpuAddr, HEX);
        return;
      }
    }
    if (erroresI2CSeguidos == 5 && corrigiendo) {
      ultimaCorreccion = 0;
      aplicarMovimiento(0);
    }
    if (erroresI2CSeguidos >= MAX_ERRORES_I2C) {
      mpuListo = false;
      ultimoReintento = ahora;
      registrarError(ERR_MPU_PAUSADO);
      apagarCorreccion();
      recuperarI2C();
      Serial.println("MPU6050 sin respuesta: se pausa la lectura y se reintenta cada 3s.");
    }
    return;
  }
  erroresI2CSeguidos = 0;

  float accelAnguloX = atan2(ay, az) * 180.0 / PI;
  float accelAnguloY = atan2(-ax, sqrt((long)ay*ay + (long)az*az)) * 180.0 / PI;

  if (primeraLectura) {
    anguloX = accelAnguloX;
    anguloY = accelAnguloY;
    anguloZ = 0;
    anguloXImpreso = anguloX;
    anguloYImpreso = anguloY;
    anguloZImpreso = anguloZ;
    tiempoAnteriorGiro = ahora;
    primeraLectura = false;
    return;
  }

  float dt = (ahora - tiempoAnteriorGiro) / 1000.0;
  tiempoAnteriorGiro = ahora;

  // Auto-ajuste lento del offset del giroscopio mientras el robot esta parado y quieto.
  static bool estabaParado = false;
  static unsigned long paradoDesde = 0;
  bool parado = (movimiento == PARADO && !frenando);
  if (parado && !estabaParado) paradoDesde = ahora;
  estabaParado = parado;
  if (parado && giroCalibrado && ahora - paradoDesde >= PARADO_MIN_AUTOCERO &&
      fabsf(gx - gxOffset) < UMBRAL_AUTOCERO && fabsf(gy - gyOffset) < UMBRAL_AUTOCERO &&
      fabsf(gz - gzOffset) < UMBRAL_AUTOCERO) {
    gxOffset += (gx - gxOffset) * FACTOR_AUTOCERO;
    gyOffset += (gy - gyOffset) * FACTOR_AUTOCERO;
    gzOffset += (gz - gzOffset) * FACTOR_AUTOCERO;
  }

  float giroVertical = (gx - gxOffset) * arribaX + (gy - gyOffset) * arribaY + (gz - gzOffset) * arribaZ;
  float velocidadAngularZ = signoGiro * giroVertical / SENSIBILIDAD_GIRO;
  if (abs(velocidadAngularZ) <= ZONA_MUERTA_GIRO) velocidadAngularZ = 0;
  velAngularZ = velocidadAngularZ;

  // El giro (rumbo) se integra SIEMPRE, tambien durante un choque; solo se descartan
  // los angulos X/Y del acelerometro cuando saltan de forma imposible.
  anguloZ += velocidadAngularZ * dt;

  bool cambioImposible = abs(diferenciaAngular(accelAnguloX, anguloX)) > CAMBIO_MAXIMO_POSIBLE ||
                         abs(diferenciaAngular(accelAnguloY, anguloY)) > CAMBIO_MAXIMO_POSIBLE ||
                         abs(gx) > 32000 || abs(gy) > 32000 || abs(gz) > 32000;
  bool aceptarAngulos = true;
  if (cambioImposible) {
    rechazosSeguidos++;
    if (rechazosSeguidos < MAX_RECHAZOS_SEGUIDOS) {
      registrarError(ERR_CAMBIO_IMPOSIBLE);
      aceptarAngulos = false;
    } else {
      rechazosSeguidos = 0;
    }
  } else {
    rechazosSeguidos = 0;
  }

  if (aceptarAngulos) {
    if (abs(diferenciaAngular(accelAnguloX, anguloX)) > UMBRAL) anguloX = accelAnguloX;
    if (abs(diferenciaAngular(accelAnguloY, anguloY)) > UMBRAL) anguloY = accelAnguloY;
  }

  if (corrigiendo) corregirRumbo(velocidadAngularZ, dt);

  if (MOSTRAR_ANGULOS && (abs(diferenciaAngular(anguloX, anguloXImpreso)) > UMBRAL ||
      abs(diferenciaAngular(anguloY, anguloYImpreso)) > UMBRAL ||
      abs(diferenciaAngular(anguloZ, anguloZImpreso)) > UMBRAL)) {

    Serial.print("X: ");
    Serial.print(anguloX);
    Serial.print("  Y: ");
    Serial.print(anguloY);
    Serial.print("  Z: ");
    Serial.println(anguloZ);

    anguloXImpreso = anguloX;
    anguloYImpreso = anguloY;
    anguloZImpreso = anguloZ;
  }
}

void comprobacionInicial() {
  int problemas = 0;

  Serial.println();
  Serial.println("===== COMPROBACION INICIAL =====");

  esp_reset_reason_t motivo = esp_reset_reason();
  Serial.print("[1] Arranque: ");
  switch (motivo) {
    case ESP_RST_POWERON:
      Serial.println("encendido normal (o recien subido el codigo)");
      break;
    case ESP_RST_SW:
      Serial.println("reinicio por software");
      break;
    case ESP_RST_EXT:
      Serial.println("reinicio con el boton EN/RESET");
      break;
    case ESP_RST_BROWNOUT:
      Serial.println("ADVERTENCIA: se reinicio por BROWNOUT (caida de voltaje)");
      registrarError(ERR_REINICIO_BROWNOUT);
      problemas++;
      break;
    case ESP_RST_PANIC:
      Serial.println("ADVERTENCIA: el programa se cayo (crash) y se reinicio");
      registrarError(ERR_REINICIO_CRASH);
      problemas++;
      break;
    case ESP_RST_INT_WDT:
    case ESP_RST_TASK_WDT:
    case ESP_RST_WDT:
      Serial.println("ADVERTENCIA: reinicio por watchdog (el programa se trabo)");
      registrarError(ERR_REINICIO_WATCHDOG);
      problemas++;
      break;
    default:
      Serial.print("otro motivo (codigo ");
      Serial.print((int)motivo);
      Serial.println(")");
      break;
  }

  Serial.print("[2] Motores/driver: pines configurados, STBY ");
  if (digitalRead(pinSTBY)) {
    Serial.println("en HIGH (driver activo)");
  } else {
    Serial.println("en LOW (driver APAGADO)");
    problemas++;
  }

  mpuListo = false;
  bool mpuPresente = iniciarMPU();

  if (!mpuPresente) {
    Serial.println("[3] MPU6050: FALLO, no responde en 0x68 ni en 0x69");
    Serial.println("    Revisa VCC (3.3V), GND, SDA (GPIO 21) y SCL (GPIO 22).");
    registrarError(ERR_MPU_NO_DETECTADO);
    problemas++;
  } else {
    delay(50);
    if (leerID() != 0x68) {
      Serial.println("[3] MPU6050: FALLO, responde pero no se identifica como MPU6050");
      registrarError(ERR_MPU_NO_DETECTADO);
      problemas++;
    } else {
      int r = leerMPU();
      if (r == 0) {
        mpuListo = true;
        float g = sqrt((float)ax * ax + (float)ay * ay + (float)az * az) / LSB_POR_G;
        Serial.print("[3] MPU6050: OK en 0x");
        Serial.print(mpuAddr, HEX);
        Serial.print(". Aceleracion total: ");
        Serial.print(g, 2);
        Serial.println(" G (cerca de 1.00 si esta quieto)");
      } else {
        registrarError(r == 1 ? ERR_I2C_LECTURA : ERR_LECTURA_VACIA);
        Serial.println("[3] MPU6050: FALLO al leer datos");
        problemas++;
      }
    }
  }

#ifdef FILTRAR_MAC
  Serial.println("[4] Mando: esperando que presiones PS en el mando autorizado (filtro de MAC activado)");
#else
  Serial.println("[4] Mando: esperando que presiones PS en el mando para conectar (sin filtro de MAC)");
#endif

  uint32_t libre = ESP.getFreeHeap();
  Serial.print("[5] Memoria libre: ");
  Serial.print(libre);
  if (libre < 40000) {
    Serial.println(" bytes, ADVERTENCIA: muy baja");
    registrarError(ERR_MEMORIA_BAJA);
    problemas++;
  } else {
    Serial.println(" bytes, OK");
  }

  Serial.print("[6] Memoria permanente: ");
  if (nvsListo) {
    Serial.print("OK, arranque #");
    Serial.print(arranqueActual);
    if (erroresPrevios > 0) {
      Serial.print(", hay ");
      Serial.print(erroresPrevios);
      Serial.println(" error(es) guardados de antes (escribe 'errores' para verlos)");
    } else {
      Serial.println(", sin errores guardados de antes");
    }
  } else {
    Serial.println("FALLO, los errores NO se van a guardar");
    problemas++;
  }

  Serial.print("[7] Calibracion del giroscopio: ");
  if (CALIBRAR_AL_ARRANCAR) {
    Serial.println("se calibra al arrancar (robot quieto y apoyado en el piso, 2 segundos)");
  } else if (!giroCalibrado) {
    Serial.println("SIN CALIBRAR, la correccion de rumbo NO actua. Escribe 'calibrar' con el robot quieto");
    problemas++;
  } else {
    Serial.print("OK, placa inclinada ");
    Serial.print(inclinacionPlaca(), 1);
    Serial.println(" grados (compensada)");
  }
  Serial.print("    Rumbo: ");
  Serial.print(rumboHabilitado ? "activado" : "desactivado");
  Serial.print(", kp=");
  Serial.print(kp, 2);
  Serial.print(", kd=");
  Serial.print(kd, 2);
  Serial.print(", ki=");
  Serial.print(ki, 2);
  Serial.print(", inversion=");
  Serial.println(signoGiro);

  if (erroresSinGuardar) guardarErrores();

  Serial.println("--------------------------------");
  if (problemas == 0) {
    Serial.println("RESULTADO: todo OK");
  } else {
    Serial.print("RESULTADO: ");
    Serial.print(problemas);
    Serial.println(" problema(s). Escribe 'errores' para ver el detalle.");
  }
  Serial.println("Comandos: errores | limpiar errores | rumbo | kp | kd | ki | signo | calibrar | ajustes | bt | pwmmin | trim | bip | volumen | temp");
  Serial.println("================================");
}

void setup() {
  Serial.begin(115200);
  delay(500);

  cargarErrores();
  cargarAjustes();

  pinMode(pinSTBY, OUTPUT);
  pinMode(pinAIN1, OUTPUT);
  pinMode(pinAIN2, OUTPUT);
  pinMode(pinBIN1, OUTPUT);
  pinMode(pinBIN2, OUTPUT);

  PWM_INICIAR(pinPWMA, canalPWMA);
  PWM_INICIAR(pinPWMB, canalPWMB);

  digitalWrite(pinSTBY, HIGH);

  BUZZER_INICIAR();
  BUZZER_ESCRIBIR(0);

  iniciarI2C();   // MPU6050: SDA=21, SCL=22, AD0 a GND

  BP32.setup(&onConnectedController, &onDisconnectedController);

  comprobacionInicial();

  if (CALIBRAR_AL_ARRANCAR) calibrarGiroscopio();

  tiempoAnteriorGiro = millis();
  ultimaLecturaGiro = millis();
  ultimoReintento = millis();
  primeraLectura = true;
  erroresI2CSeguidos = 0;
  rechazosSeguidos = 0;
}

void loop() {
  unsigned long ahora = millis();

  static String bufferSerial = "";
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\r') continue;
    if (c == '\n') {
      procesarComandoSerial(bufferSerial);
      bufferSerial = "";
    } else {
      if (bufferSerial.length() < 64) bufferSerial += c;   // limite: evita crecer sin fin
    }
  }

  BP32.update();

  leerMando();
  mostrarBotones();
  actualizarMovimiento();
  actualizarMPU(ahora);
  actualizarBuzzer(ahora);

  // Escribir en flash puede frenar el loop unas decenas de ms: solo con el robot parado,
  // salvo que lleve mucho sin guardarse.
  if (erroresSinGuardar) {
    bool quieto = (movimiento == PARADO && !frenando);
    unsigned long desdeGuardado = ahora - ultimoGuardado;
    if ((quieto && desdeGuardado >= INTERVALO_GUARDADO) || desdeGuardado >= 4 * INTERVALO_GUARDADO) {
      guardarErrores();
    }
  }

  vTaskDelay(1);   // cede tiempo al Bluetooth; sin esto el mando llega con retraso
}
